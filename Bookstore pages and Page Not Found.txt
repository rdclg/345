Bookstore pages and Page Not Found
import { BrowserRouter, Routes, Route, Link } from "react-router-dom";
function Home() { return <h2>Home</h2>; }
function Books() { return <h2>Books</h2>; }
function Cart() { return <h2>Cart</h2>; }
function NotFound() { return <h2>Page Not Found</h2>; }
export default function App() {
  return <BrowserRouter>
    <Link to="/">Home</Link> | <Link to="/books">Books</Link> | <Link to="/cart">Cart</Link>
    <Routes>
      <Route path="/" element={<Home />} />
      <Route path="/books" element={<Books />} />
      <Route path="/cart" element={<Cart />} />
      <Route path="*" element={<NotFound />} />
    </Routes>
  </BrowserRouter>;
}
