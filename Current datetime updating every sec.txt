 Current date/time updating every second
import { useState, useEffect } from "react";
function App() {
  const [time, setTime] = useState(new Date());
  useEffect(() => {
    const id = setInterval(() => setTime(new Date()), 1000);
    return () => clearInterval(id);
  }, []);
  return <h2>{time.toLocaleString()}</h2>;
}
export default App;
Expected output:
Current date and time, updating each second
Remember: useEffect runs the timer setup.
