Connect Node.js using Mongoose and insert student
// Install: npm install mongoose
const mongoose = require("mongoose");
mongoose.connect("mongodb://127.0.0.1:27017/CollegeDB")
  .then(async () => {
    const Student = mongoose.model("Student",
      new mongoose.Schema({ name: String, marks: Number }));
    const s = await Student.create({ name: "Aman", marks: 80 });
    console.log(s);
    await mongoose.disconnect();
  })
  .catch(err => console.log(err));
Run: node server.js
Expected output:
Console shows the inserted document
